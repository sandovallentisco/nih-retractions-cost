pubmed_funding_project/
|
+-- data/
|   +-- raw/                                  # Unprocessed source data (large; not committed)
|   |   +-- nih_reporter/                     # NIH ExPORTER CSVs, one file per fiscal year
|   |   |   +-- RePORTER_PRJ_C_FY####.csv        # Project files: PI names, PI_IDS, ORG_NAME, activity code
|   |   |   +-- RePORTER_PRJFUNDING_C_FY####.csv # Yearly funding-amount files
|   |   +-- retraction_watch/                 # Manual GitLab download of the Retraction Watch Database
|   |
|   +-- processed/                            # Generated by the Python pipeline / R scripts; read by the manuscripts and the app
|       +-- MASTER_NIH_Projects.csv           # NIH projects compressed to a total cost per grant
|       +-- MASTER_PI_History.csv             # Complete yearly funding history of each PI/grant
|       +-- FINAL_Retractions_Costs_and_Pubs.csv  # Paper-level table: Retraction Watch merged with funding + publication counts
|       +-- Author_Funding_Matches.csv        # Retracted author -> grant / PI / fiscal-year funding (consequences timeline)
|       +-- annual_cpi.csv                    # Yearly CPI and inflation multipliers (from src/cpi_fetcher.py)
|       +-- author_collision_flags.csv        # Per-author name-collision flags        (built by build_collision_flags.R)
|       +-- author_grant_keep.csv             # Per (author, grant) keep flag for the correction (built by build_collision_flags.R)
|       +-- (intermediate CSVs: retraction_watch_with_funding.csv,
|           FINAL_Retractions_with_Costs.csv, pubmed_cache.csv)
|
+-- src/                                      # Python data pipeline (steps 1-6)
|   +-- config.py                             # Global paths and API credentials (read from .env)
|   +-- downloader.py                         # Downloads the raw NIH ExPORTER and Retraction Watch data
|   +-- data_handler.py                       # Filters U.S. papers and removes publisher-error retractions
|   +-- entrez_client.py                      # Queries PubMed (NCBI Entrez) for grants and study designs
|   +-- pubmed_cache.py                       # Local cache of PubMed responses (avoids re-querying)
|   +-- seed_cache_from_existing.py           # Seeds the PubMed cache from previously retrieved results
|   +-- cpi_fetcher.py                        # Fetches the CPI series (FRED) -> annual_cpi.csv
|   +-- pipeline.py                           # Orchestrates step 1 (PubMed metadata extraction)
|   +-- nih_merger.py                         # Aggregates millions of NIH rows into total grant costs (step 2)
|   +-- pi_history_generator.py               # Builds the yearly PI/grant history (step 3)
|   +-- funding_cleaner_linker.py             # Matches PubMed grants with NIH dollars (step 4)
|   +-- publication_counter.py                # Counts how many papers each grant produced (step 5)
|   +-- author_funding_matcher.py             # Computes diff_retracted for the consequences analysis (step 6)
|
+-- main.py                                   # Interactive console menu to run pipeline steps 1-6 manually
|
+-- build_collision_flags.R                   # Detects retracted-author name collisions from the raw ExPORTER files and
|                                             #   writes author_collision_flags.csv + author_grant_keep.csv.
|                                             #   Run once after major data updates (needs data/raw/nih_reporter).
|
+-- preprint.qmd                              # MAIN MANUSCRIPT (Quarto): abstract, direct cost, author consequences, Figures 1-2
+-- supplementary_materials.qmd               # Supplementary methods, results and figures (Quarto)
+-- references.bib                            # Bibliography for both manuscripts
+-- science.csl                               # Citation style (numbered, Science) for both manuscripts
|                                             #   (both .qmd render to .html / .pdf / .docx)
|
+-- app.R                                     # Shiny dashboard: reads processed CSVs, applies inflation, computes stats, renders the site
+-- deploy_shiny.R                            # Publishes app.R + the CSVs it needs to shinyapps.io
|
+-- .github/
|   +-- workflows/
|       +-- weekly_update.yml                 # GitHub Actions: weekly automated data refresh + dashboard redeploy ("living analysis")
|
+-- requirements.txt                          # Python libraries the pipeline / CI need to install
+-- README.md                                 # Project overview
+-- SETUP.md                                  # Setup and how-to-run instructions
+-- LICENSE
+-- .env                                      # ENTREZ_EMAIL and NCBI API_KEY (not committed)
