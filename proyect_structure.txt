📁 pubmed_funding_project/
├── 📁 data/
│   ├── 📁 raw/                           # Raw material (Unprocessed data)
│   │   ├── 📁 nih_reporter/              # CSVs downloaded from NIH (RePORTER_PRJ_C_FY...).
│   │       ├── RePORTER_PRJ_C_FY1985.csv  
│   │       ├── RePORTER_PRJ_C_FY1986.csv
│   │       └── ...
│   │   └── 📁 retraction_watch/          # Where the manual GitLab CSV download was stored.
│   │
│   └── 📁 processed/                     # Generated by Python, read by R
│       ├── 📄 MASTER_NIH_Projects.csv    # Compressed NIH database with total costs per Grant.
│       ├── 📄 MASTER_PI_History.csv      # Complete yearly history of each author and their active projects.
│       ├── 📄 FINAL_Retractions...csv    # Retraction Watch papers merged with funding and publication counts.
│       └── 📄 Author_Funding_Matches.csv # The survival "timeline" (funding pre- and post-retraction).
│
├── 📁 src/                               # Specialized modules
│   ├── 📄 config.py                      # Global paths and API credentials setup (.env).
│   ├── 📄 data_handler.py                # Filters US papers and removes publisher-error retractions.
│   ├── 📄 entrez_client.py               # Connects to PubMed to extract study designs and Grants.
│   ├── 📄 pipeline.py                    # Orchestrates Step 1 (PubMed metadata extraction with progress bar).
│   ├── 📄 nih_merger.py                  # Aggregates millions of NIH rows into total grant costs (Step 2).
│   ├── 📄 pi_history_generator.py        # Builds the yearly history for authors (Step 3).
│   ├── 📄 funding_cleaner_linker.py      # Matches PubMed Grants with NIH dollars (Step 4).
│   ├── 📄 publication_counter.py         # Counts how many papers each Grant has produced (Step 5).
│   └── 📄 author_funding_matcher.py      # Calculates the year difference (diff_retracted) for survival analysis (Step 6).
│
├── 📄 main.py                            # Interactive console menu to run steps 1-6 manually.
│
├── 📄 app.R                      	    # ShinyApp R script. Reads 'processed' CSVs, applies FRED inflation, calculates stats, and renders the website.
│
├── 📄 requirements.txt                   # Tells the GitHub server which Python libraries to install (pandas, biopython, requests...).
│
└── 📄 .env                               # File containing ENTREZ_EMAIL and API_KEY.
